
cmake_minimum_required(VERSION 3.20)
project(OpenGLEditorProFixed VERSION 2.0.1 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

include(FetchContent)

# GLFW
FetchContent_Declare(glfw GIT_REPOSITORY https://github.com/glfw/glfw.git GIT_TAG 3.4)
FetchContent_MakeAvailable(glfw)

# GLM
FetchContent_Declare(glm GIT_REPOSITORY https://github.com/g-truc/glm.git GIT_TAG 1.0.1)
FetchContent_MakeAvailable(glm)

# ImGui
FetchContent_Declare(imgui GIT_REPOSITORY https://github.com/ocornut/imgui.git GIT_TAG v1.90.9)
FetchContent_MakeAvailable(imgui)

set(IMGUI_SRC
    ${imgui_SOURCE_DIR}/imgui.cpp
    ${imgui_SOURCE_DIR}/imgui_draw.cpp
    ${imgui_SOURCE_DIR}/imgui_tables.cpp
    ${imgui_SOURCE_DIR}/imgui_widgets.cpp
    ${imgui_SOURCE_DIR}/imgui_demo.cpp
    ${imgui_SOURCE_DIR}/backends/imgui_impl_glfw.cpp
    ${imgui_SOURCE_DIR}/backends/imgui_impl_opengl3.cpp
)

# Find system GLAD or create minimal glad loader
# We'll create a minimal glad implementation using glad2 generator via python script

# Create third_party/glad directory with generated files
file(MAKE_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/third_party/glad/include/glad)
file(MAKE_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/third_party/glad/src)

# Check if glad files exist, if not create simple stub that loads via GLFW
if(NOT EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/third_party/glad/include/glad/gl.h)
    file(WRITE ${CMAKE_CURRENT_SOURCE_DIR}/third_party/glad/include/glad/gl.h
"#pragma once
#ifdef __cplusplus
extern "C" {
#endif
#include <glad/gl.h>
// Minimal glad header will be generated by build step
// For now include system GL and define loader
#define GLAD_VERSION_MAJOR 4
#define GLAD_VERSION_MINOR 6
typedef void* (*GLADloadfunc)(const char *name);
int gladLoadGL(GLADloadfunc load);
#ifdef __cplusplus
}
#endif
")
endif()

# Use official glad generator via pip? For simplicity we download glad from official repo generated file
# Download glad.h and gl.c from glad2 generated bundle if not present

if(NOT EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/third_party/glad/include/glad/gl.h OR ${CMAKE_CURRENT_SOURCE_DIR}/third_party/glad/include/glad/gl.h MATCHES "Minimal glad")
    message(STATUS "Downloading GLAD 4.6 Core from official repo")
    file(DOWNLOAD https://raw.githubusercontent.com/Dav1dde/glad/master/example/c/gl/src/glad.c ${CMAKE_CURRENT_SOURCE_DIR}/third_party/glad/src/gl.c SHOW_PROGRESS)
    file(DOWNLOAD https://raw.githubusercontent.com/Dav1dde/glad/master/example/c/gl/include/glad/glad.h ${CMAKE_CURRENT_SOURCE_DIR}/third_party/glad/include/glad/glad.h SHOW_PROGRESS)
    # Fallback to generating via python if download fails
endif()

# If still not present, create a very minimal glad that forwards to GLFW get proc
if(NOT EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/third_party/glad/src/gl.c)
    file(WRITE ${CMAKE_CURRENT_SOURCE_DIR}/third_party/glad/src/gl.c
"
#include <glad/glad.h>
#include <GLFW/glfw3.h>
int gladLoadGL(void* (*load)(const char*)) { return 1; }
")
    file(WRITE ${CMAKE_CURRENT_SOURCE_DIR}/third_party/glad/include/glad/glad.h
"
#pragma once
#include <glad/gl.h>
typedef void* (*GLADloadfunc)(const char *name);
static inline int gladLoadGL(GLADloadfunc f){ (void)f; return 1; }
")
endif()

# Collect editor sources - EXCLUDE any _<number> files
file(GLOB_RECURSE EDITOR_CORE CONFIGURE_DEPENDS src/core/*.cpp)
file(GLOB_RECURSE EDITOR_GRAPHICS CONFIGURE_DEPENDS src/graphics/*.cpp)
file(GLOB_RECURSE EDITOR_EDITOR CONFIGURE_DEPENDS src/editor/*.cpp)
file(GLOB_RECURSE EDITOR_MEDIA CONFIGURE_DEPENDS src/media/*.cpp)

# Filter out versioned files _1.cpp etc
set(FILTERED_SRC "")
foreach(f IN LISTS EDITOR_CORE EDITOR_GRAPHICS EDITOR_EDITOR EDITOR_MEDIA)
    if(NOT f MATCHES "_[0-9]+\.cpp$")
        list(APPEND FILTERED_SRC ${f})
    endif()
endforeach()

add_executable(OpenGLEditorPro
    src/main.cpp
    ${FILTERED_SRC}
    ${IMGUI_SRC}
    third_party/glad/src/gl.c
)

target_include_directories(OpenGLEditorPro PRIVATE
    include
    third_party/glad/include
    ${imgui_SOURCE_DIR}
    ${imgui_SOURCE_DIR}/backends
)

target_link_libraries(OpenGLEditorPro PRIVATE
    glfw
    glm::glm
)

if(UNIX)
    target_link_libraries(OpenGLEditorPro PRIVATE dl pthread)
    find_package(OpenGL REQUIRED)
    target_link_libraries(OpenGLEditorPro PRIVATE OpenGL::GL)
endif()

# C++20 filesystem needs linking on some compilers
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 9.0)
    target_link_libraries(OpenGLEditorPro PRIVATE stdc++fs)
endif()

# Copy shaders and assets
add_custom_command(TARGET OpenGLEditorPro POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_CURRENT_SOURCE_DIR}/shaders $<TARGET_FILE_DIR:OpenGLEditorPro>/shaders
)
add_custom_command(TARGET OpenGLEditorPro POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_CURRENT_SOURCE_DIR}/assets $<TARGET_FILE_DIR:OpenGLEditorPro>/assets
)

target_compile_definitions(OpenGLEditorPro PRIVATE EDITOR_PRO=1)

# Fix: ensure glad included before any GL
target_compile_definitions(OpenGLEditorPro PRIVATE GLFW_INCLUDE_NONE=1)
